\section*{Q1: Cart Pendulum Modelling \hfill (20 marks)}
%=======================================================================

Using the required generalised coordinates,
\begin{equation*}
	q_{1} = x_{c} \qquad\qquad q_{2} = \phi
\end{equation*}
where $x_{c}$ is the horizontal displacement of the cart and $\phi$ is measured
anticlockwise from the negative $x$-axis.

From the geometry of the system, the pendulum mass position is
\begin{align*}
	x_{p} &= x_{c} - l\cos\phi \\
	y_{p} &= -l\sin\phi
\end{align*}
where $l$ is the distance from the pivot to the centre of oscillation.

Differentiating with respect to time,
\begin{align*}
	\dot{x}_{p} &= \dot{x}_{c} + l\dot{\phi}\sin\phi \\
	\dot{y}_{p} &= -l\dot{\phi}\cos\phi
\end{align*}

The total co-kinetic energy of the system is the sum of the cart and pendulum
kinetic energies:
\begin{equation*}
	T^{*} = T_{c} + T_{p}
\end{equation*}
with
\begin{equation*}
	T_{c} = \tfrac{1}{2} M \dot{x}_{c}^{2}
\end{equation*}
and
\begin{equation*}
	T_{p} = \tfrac{1}{2} m \left( \dot{x}_{p}^{2} + \dot{y}_{p}^{2} \right)
\end{equation*}

Therefore,
\begin{equation*}
	T^{*} = \frac{1}{2} M \dot{x}_{c}^{2}
	+ \frac{1}{2} m \left[ \left( \dot{x}_{c} + l\dot{\phi}\sin\phi \right)^{2}
	+ \left( -l\dot{\phi}\cos\phi \right)^{2} \right]
\end{equation*}

Simplifying,
\begin{align*}
	T^{*} &= \frac{1}{2} M \dot{x}_{c}^{2}
	+ \frac{1}{2} m \left( \dot{x}_{c}^{2}
	+ 2 l \dot{x}_{c} \dot{\phi} \sin\phi + l^{2}\dot{\phi}^{2} \right) \\
	T^{*} &= \frac{1}{2} \left( M + m \right) \dot{x}_{c}^{2}
	+ m l \dot{x}_{c} \dot{\phi} \sin\phi
	+ \frac{1}{2} m l^{2} \dot{\phi}^{2}
\end{align*}

Taking the pivot height as the gravitational reference,
\begin{align*}
	V &= m g\, y_{p} \\
	V &= -m g l \sin\phi
\end{align*}

The viscous friction of the cart and pendulum is represented using the Rayleigh
dissipation function,
\begin{equation*}
	D = \frac{1}{2} b_{c} \dot{x}_{c}^{2} + \frac{1}{2} b_{p} \dot{\phi}^{2}
\end{equation*}

The Lagrangian of the system is defined as
\begin{equation*}
	L = T^{*} - V
\end{equation*}

Substituting the co-kinetic and potential energy expressions gives
\begin{equation*}
	L = \frac{1}{2} \left( M + m \right) \dot{x}_{c}^{2}
	+ m l \dot{x}_{c} \dot{\phi} \sin\phi
	+ \frac{1}{2} m l^{2} \dot{\phi}^{2}
	+ m g l \sin\phi
\end{equation*}

The Euler--Lagrange equation including viscous dissipation is
\begin{equation*}
	\frac{d}{dt} \left( \frac{\partial L}{\partial \dot{q}_{i}} \right)
	- \frac{\partial L}{\partial q_{i}}
	+ \frac{\partial D}{\partial \dot{q}_{i}} = Q_{i}
\end{equation*}

For $q_{1} = x_{c}$,
\begin{equation*}
	\frac{\partial L}{\partial \dot{x}_{c}}
	= \left( M + m \right) \dot{x}_{c} + m l \dot{\phi} \sin\phi
\end{equation*}

Taking the time derivative,
\begin{equation*}
	\frac{d}{dt} \left( \frac{\partial L}{\partial \dot{x}_{c}} \right)
	= \left( M + m \right) \ddot{x}_{c}
	+ m l \ddot{\phi} \sin\phi
	+ m l \dot{\phi}^{2} \cos\phi
\end{equation*}

Also,
\begin{equation*}
	\frac{\partial L}{\partial x_{c}} = 0
\end{equation*}
and
\begin{equation*}
	\frac{\partial D}{\partial \dot{x}_{c}} = b_{c} \dot{x}_{c}
\end{equation*}

Since the generalised force associated with $x_{c}$ is
\begin{equation*}
	Q_{x_{c}} = F
\end{equation*}
substitution into the Euler--Lagrange equation gives the cart equation
\begin{equation*}
	\boxed{\;
		F = \left( M + m \right) \ddot{x}_{c}
		+ b_{c} \dot{x}_{c}
		+ m l \ddot{\phi} \sin\phi
		+ m l \dot{\phi}^{2} \cos\phi
		\;} \tag{1a}
\end{equation*}

Using the same Euler--Lagrange equation:
\begin{equation*}
	\frac{d}{dt} \left( \frac{\partial L}{\partial \dot{\phi}} \right)
	- \frac{\partial L}{\partial \phi}
	+ \frac{\partial D}{\partial \dot{\phi}} = Q_{\phi}
\end{equation*}

For this system,
\begin{equation*}
	Q_{\phi} = 0
\end{equation*}

Considering the pendulum coordinate $q_{2} = \phi$,
\begin{equation*}
	\frac{\partial L}{\partial \dot{\phi}}
	= m l \dot{x}_{c} \sin\phi + m l^{2} \dot{\phi}
\end{equation*}

Taking the time derivative,
\begin{equation*}
	\frac{d}{dt} \left( \frac{\partial L}{\partial \dot{\phi}} \right)
	= m l \ddot{x}_{c} \sin\phi
	+ m l \dot{x}_{c} \dot{\phi} \cos\phi
	+ m l^{2} \ddot{\phi}
\end{equation*}

Now differentiate $L$ with respect to $\phi$:
\begin{equation*}
	\frac{\partial L}{\partial \phi}
	= m l \dot{x}_{c} \dot{\phi} \cos\phi + m g l \cos\phi
\end{equation*}

For the dissipation term,
\begin{equation*}
	\frac{\partial D}{\partial \dot{\phi}} = b_{p} \dot{\phi}
\end{equation*}

Substituting into Euler--Lagrange:
\begin{equation*}
	m l \ddot{x}_{c} \sin\phi
	+ m l \dot{x}_{c} \dot{\phi} \cos\phi
	+ m l^{2} \ddot{\phi}
	- \left( m l \dot{x}_{c} \dot{\phi} \cos\phi + m g l \cos\phi \right)
	+ b_{p} \dot{\phi} = 0
\end{equation*}

The matching cross terms cancel, leaving
\begin{equation*}
	\boxed{\;
		m l^{2} \ddot{\phi} + b_{p} \dot{\phi}
		+ m l \ddot{x}_{c} \sin\phi - m g l \cos\phi = 0
		\;} \tag{1b}
\end{equation*}

\section*{Q2: Analytical Linear Model of a Cart Pendulum System
	\hfill (40 marks)}
%=======================================================================

The nonlinear model provided:
\begin{align}
	K_{v} \cdot u &= \left( M + m \right) \ddot{x}_{c} + b_{c} \dot{x}_{c}
	+ m \ell \ddot{\alpha} \cos\alpha
	- m \ell \dot{\alpha}^{2} \sin\alpha \tag{1a} \\
	0 &= m \ell^{2} \ddot{\alpha} + b_{p} \dot{\alpha}
	+ m \ell \ddot{x}_{c} \cos\alpha + g m \ell \sin\alpha \tag{1b}
\end{align}

\subsection*{Q2.1)}

For this question,

The system is linearised around $\alpha = 0$ corresponding to the pendulum
vertically facing downwards.

For small angular displacements around $\alpha = 0$,
\begin{align*}
	\sin\alpha &\approx \alpha \\
	\cos\alpha &\approx 1
\end{align*}

Non-linear higher-order term:
\begin{equation*}
	\dot{\alpha}^{2} \sin\alpha \approx 0
\end{equation*}

Applying approximations to (1a):
\begin{equation*}
	K_{v} u = \left( M + m \right) \ddot{x}_{c} + b_{c} \dot{x}_{c}
	+ m l \ddot{\alpha}(1) - m l (0)
\end{equation*}

Applying approximations to (1b):
\begin{equation*}
	0 = m l^{2} \ddot{\alpha} + b_{p} \dot{\alpha}
	+ m l \ddot{x}_{c}(1) + g m l (\alpha)
\end{equation*}

Therefore,
\begin{align}
	K_{v} u &= \left( M + m \right) \ddot{x}_{c} + b_{c} \dot{x}_{c}
	+ m \ell \ddot{\alpha} \tag{2a} \\
	0 &= m \ell^{2} \ddot{\alpha} + b_{p} \dot{\alpha}
	+ m \ell \ddot{x}_{c} + g m \ell \alpha \tag{2b}
\end{align}

\subsection*{Q2.2)}

Assuming zero initial conditions, applying Laplace transform to (2a):
\begin{equation*}
	K_{v} U(s) = \left( M + m \right) s^{2} X_{c}(s) + b_{c} s X_{c}(s)
	+ m l s^{2} A(s)
\end{equation*}

Taking Laplace transform of (2b):
\begin{equation*}
	0 = m l^{2} s^{2} A(s) + b_{p} s A(s) + m l s^{2} X(s) + g m l A(s)
\end{equation*}

\begin{align}
	K_{v} U(s) &= \left[ \left( M + m \right) s^{2} + b_{c} s \right] X_{c}(s)
	+ m \ell s^{2} A(s) \tag{3a} \\
	0 &= m \ell s^{2} X(s)
	+ \left[ m \ell^{2} s^{2} + b_{p} s + g m \ell \right] A(s) \tag{3b}
\end{align}

$G_{x\alpha}(s)$:

Rearranging (3b),
\begin{equation*}
	\left[ m l^{2} s^{2} + b_{p} s + g m l \right] A(s) = -m l s^{2} X(s)
\end{equation*}

\begin{equation*}
	\frac{A(s)}{X(s)} = \frac{-m l s^{2}}{m l^{2} s^{2} + b_{p} s + g m l}
\end{equation*}

\begin{align}
	G_{x\alpha}(s) &= \frac{A(s)}{X(s)}
	= -\frac{m l s^{2}}{m l^{2} s^{2} + b_{p} s + g m l} \tag{4a} \\
	G_{x\alpha}(s) &= -\frac{s^{2}}{l s^{2} + \dfrac{b_{p}}{m l} s + g} \tag{4b}
\end{align}

\subsection*{Q3.1) Identification of $\tilde G_{ux}(s)$}

Following Remark~2, the identification is carried out on the
\emph{voltage-to-speed} channel and the integrator is appended afterwards,
which avoids fitting the marginally stable pole at $s=0$.

From~\eqref{eq:Guvc_simple} the model structure to be fitted is first
order with no zeros, so in \texttt{main\_Q31\_Q32.m} the input/output
signals and model orders must be set to
\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
\texttt{y  = vc;}   & TF output -- filtered derivative of \texttt{xc\_m} \\
\texttt{u  = Vm;}   & TF input  -- measured motor voltage \\
\texttt{np = 1;}    & one pole (from $s+b$) \\
\texttt{nz = 0;}    & no zeros \\
\bottomrule
\end{tabular}
\end{center}
so that \texttt{Guvc\_id} $=\hat a/(s+\hat b)$ and
\texttt{Gux\_id} $=\tfrac1s$\texttt{Guvc\_id}. Record the fitted
coefficients as
\begin{equation}
\hat G_{uv_c}(s) = \frac{\hat a}{s+\hat b}
\qquad\Longrightarrow\qquad
\tilde G_{ux}(s) = \frac{\hat a}{s(s+\hat b)} .
\label{eq:id_uvc}
\end{equation}

\emph{Numerical result (from \texttt{main\_Q31\_Q32.m}):}
$\hat a = 0.887$, $\hat b = 16.25$,
fit to estimation data $=93.97\,\%$.

\subsection*{Q3.2) Identification of $G_{x\alpha}(s)$}

Following Remark~3, the fit is performed between the cart acceleration
$a_c(t)=\ddd x_c(t)$ and the angle $\alpha(t)$. From~\eqref{eq:Gaca} the
correct structure is second order with no zeros:
\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
\texttt{y  = alpha\_rad;} & TF output -- measured angle in radians \\
\texttt{u  = ac;}         & TF input  -- filtered cart acceleration \\
\texttt{np = 2;}          & two poles (pendulum resonance) \\
\texttt{nz = 0;}          & no zeros \\
\bottomrule
\end{tabular}
\end{center}
Normalising the estimate returned by \texttt{tfest} so that the leading
denominator coefficient is unity,
\begin{equation}
\hat G_{a_c\alpha}(s)
= \frac{\hat k}{s^{2}+\hat a_1 s+\hat a_0},
\qquad\Longrightarrow\qquad
\hat G_{x\alpha}(s)=s^{2}\hat G_{a_c\alpha}(s)
= \frac{\hat k\, s^{2}}{s^{2}+\hat a_1 s+\hat a_0}.
\label{eq:id_aca}
\end{equation}

\emph{Numerical result (from \texttt{main\_Q31\_Q32.m}):}
$\hat k = -3.902$, $\hat a_1 = 0.5832$,
$\hat a_0 = 38.58$,
fit to estimation data $=78.4\,\%$.

\subsection*{Q3.3) Estimation of $b_c$, $b_p$, $\l$ and $K_v$}

The unknown parameters follow from a coefficient-by-coefficient comparison
of the identified transfer functions with the analytical ones derived in
Q2.

\paragraph{Step 1 -- from $G_{x\alpha}(s)$: obtain $\l$ and $b_p$
(Remark~4).}
Comparing~\eqref{eq:id_aca} with the analytical
expression~\eqref{eq:Gaca},
\begin{equation}
\frac{\hat k}{s^{2}+\hat a_1 s+\hat a_0}
\;\equiv\;
\frac{-1/\l}{s^{2}+\dfrac{b_p}{m\l^{2}}s+\dfrac{g}{\l}}
\qquad\Longrightarrow\qquad
\left\{
\begin{aligned}
\hat a_0 &= \frac{g}{\l},\\[3pt]
\hat a_1 &= \frac{b_p}{m\l^{2}},\\[3pt]
\hat k   &= -\frac{1}{\l}.
\end{aligned}
\right.
\end{equation}
The first relation gives the length directly, the second the pivot
friction:
\begin{empheq}[box=\fbox]{align}
\l   &= \frac{g}{\hat a_0}
        \;=\; \frac{g}{\hat\omega_n^{\,2}},
        \qquad \hat\omega_n = \sqrt{\hat a_0}
\label{eq:ell}\\[4pt]
b_p  &= m\,\l^{2}\,\hat a_1
        \;=\; \frac{m\,g^{2}\,\hat a_1}{\hat a_0^{\,2}}
        \;=\; 2\zeta\hat\omega_n\, m\,\l^{2},
        \qquad \zeta=\frac{\hat a_1}{2\sqrt{\hat a_0}}
\label{eq:bp}\\[4pt]
L    &= \frac{3}{2}\,\l \qquad\text{(since } \l = 2L/3\text{)}
\label{eq:Lrod}
\end{empheq}
The third relation, $\hat k=-1/\l$, is redundant and provides a useful
\textbf{consistency check}: the value $-1/\hat k$ should agree with
$g/\hat a_0$ to within the identification error. (If \texttt{tfest} returns
$\hat k>0$, the encoder sign convention for $\alpha$ is inverted with
respect to Fig.~2; negate \texttt{alpha\_rad} and re-identify.)

\paragraph{Step 2 -- from $\tilde G_{ux}(s)$: obtain $K_v$ and $b_c$
(Remark~4).}
Comparing~\eqref{eq:id_uvc} with the analytical
form~\eqref{eq:Guvc_simple}, $a = K_v/(M+m)$ and $b = b_c/(M+m)$, hence
\begin{empheq}[box=\fbox]{align}
K_v &= (M+m)\,\hat a
     = 0.32\,\hat a \quad [\mathrm{N/V}]
\label{eq:Kv}\\[3pt]
b_c &= (M+m)\,\hat b
     = 0.32\,\hat b \quad [\mathrm{N\,s/m}]
\label{eq:bc}
\end{empheq}
Equivalently, in terms of the steady-state gain and time constant of the
first-order speed response, $K_v = b_c\,\hat a/\hat b$ and
$b_c = (M+m)/\hat\tau$.

\subsection*{Q3.4) Nonlinear and linear simulation models}

The Simulink file \texttt{sim\_2026\_cart\_pend.slx} implements the two
models below with the parameters identified in Q3.3.

\paragraph{Nonlinear model.} Writing~(1a)--(1b) in matrix form,
\begin{equation}
\begin{bmatrix}
M+m & m\l\cos\alpha\\
m\l\cos\alpha & m\l^{2}
\end{bmatrix}
\begin{bmatrix}\ddd x_c\\ \ddd\alpha\end{bmatrix}
=
\begin{bmatrix}
K_v u - b_c\dd x_c + m\l\dd\alpha^{\,2}\sin\alpha\\
-\,b_p\dd\alpha - m g \l\sin\alpha
\end{bmatrix},
\end{equation}
whose mass-matrix determinant is
$\det = m\l^{2}\big(M+m\sin^{2}\alpha\big) > 0$ for all $\alpha$. Inverting,
\begin{align}
\ddd{x}_c &=
\frac{\l\big(K_v u - b_c\dd x_c + m\l\dd\alpha^{2}\sin\alpha\big)
      + \cos\alpha\big(b_p\dd\alpha + m g \l\sin\alpha\big)}
     {\l\big(M+m\sin^{2}\alpha\big)},
\label{eq:nl1}\\[4pt]
\ddd{\alpha} &=
\frac{-(M+m)\big(b_p\dd\alpha + m g\l\sin\alpha\big)
      - m\l\cos\alpha\big(K_v u - b_c\dd x_c
        + m\l\dd\alpha^{2}\sin\alpha\big)}
     {m\l^{2}\big(M+m\sin^{2}\alpha\big)} .
\label{eq:nl2}
\end{align}
Equations~\eqref{eq:nl1}--\eqref{eq:nl2} are implemented with integrator
blocks initialised at $x_c(0)=v_c(0)=\alpha(0)=\omega(0)=0$.

\paragraph{Linear model.} As required by Remark~6, the linear branch uses
the \textbf{full} $G_{ux}(s)$ of~\eqref{eq:Gux}--\eqref{eq:Delta} --- not
the simplified $\tilde G_{ux}(s)$ --- cascaded with $G_{x\alpha}(s)$
of~\eqref{eq:Gxa}, exactly as in Fig.~3.

\paragraph{Comparison.} \emph{[Insert the plot generated at the end of
\texttt{main\_Q33\_Q34.m} comparing $x_c(t)$ and $\alpha(t)$ from the
nonlinear and linear models for the same input $u(t)$.]}

The two responses are expected to be indistinguishable while

$\sin\alpha$ become significant --- which quantifies the validity range of
